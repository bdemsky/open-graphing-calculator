# Home screen: arithmetic, precedence, Ans, exact/approx, errors, entry recall
expect screen home
type "2+3*4\n"
expect state last_output 14
expect text 14
# operator first uses Ans: Ans*2 = 28
type "*2\n"
expect state last_input Ans*2
expect state last_output 28
type "1/3\n"
expect state last_output 1/3
key SQRT 2 RPAREN ENTER
expect state last_input √(2)
# the stub is numeric; the real CAS keeps exact roots
ifcas stub expect state last_output 1.41421356237
ifcas stub expect text 1.41421356237
ifcas src expect state last_output √(2)
ifcas src expect text √(2)
key SIN SHIFT X DIV 6 RPAREN ENTER
expect state last_output 1/2
# store and recall a variable
type "5"
key ASSIGN
type "a\n"
type "a^2\n"
expect state last_output 25
# errors are shown, not fatal (the real CAS treats q symbolically: use 1/0)
type "1/0\n"
expect state last_error 1
expect statecontains last_output ERR
# SHIFT ENTER (entry) recalls the previous entry into the editor
key SHIFT ENTER
expect state entry 1/0
key CLEAR
expect state entry ""
key 2 SHIFT X
expect state entry 2π
