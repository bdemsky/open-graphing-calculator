# Notebooks and the computer (app/usb_app.c, core/vdrive.c, modules/python/nb_doc.c; help page
# "notebook-files"): the CALC drive's NOTEBOOKS folder lists the stored notebooks as .ipynb files;
# notebooks dropped on the drive (macOS and Windows write orders, the root or NOTEBOOKS) are imported:
# one written by Jupyter (outputs, an image, an ANSI traceback, a raw cell), one by hand (string
# sources, no ids, an unknown cell type, the calculator's math and graph cells, \u escapes), a broken
# file and an nbformat 3 file (refused with the reason), a replacement (the previous version kept), an
# import waiting while the app is open, and the drive refreshed after a change.
requirecfg MOD_PYTHON
nb store wipe
nb new Algebra
nb set 1 "integrate(x^2,x)"
nb add python "print('hi')\n2**10"
nb add graph "sin(x), -5, 5"
nb runall
nb close
usb host on
usb ls NOTEBOOKS
usb expect NOTEBOOKS/Algebra.ipynb "\"nbformat\": 4,"
usb expect NOTEBOOKS/Algebra.ipynb "\"nbformat_minor\": 5"
usb expect NOTEBOOKS/Algebra.ipynb "calc.eval(r"
usb expect NOTEBOOKS/Algebra.ipynb "\"kind\": \"math\""
usb expect NOTEBOOKS/Algebra.ipynb "\"image/png\": \"iVBORw0KGgo"
usb expect NOTEBOOKS/algebra.IPYNB "\"text/plain\": "
usb expect README.TXT NOTEBOOKS
usb get NOTEBOOKS/Algebra.ipynb build/ui_nb_algebra.ipynb
# a notebook written by Jupyter, dropped in the root (macOS)
usb drop "tests/notebooks/From Jupyter.ipynb" mac
wait 1000
expect state usb_result "Imported From Jupyter"
expect state toast "Imported From Jupyter"
wait 2000
usb expect RESULT.TXT "The notebook was imported."
usb expect RESULT.TXT "6 cells"
usb expect RESULT.TXT "images are not shown"
usb expect "NOTEBOOKS/From Jupyter.ipynb" "Imported from Jupyter"
nb open "From Jupyter"
nb expect 1 src "# Imported from Jupyter"
nb expect 2 out "radius 2"
nb expect 2 val 12.566370614359172
nb expect 2 src "math.pi * 2**2"
nb expect 3 out ""
nb expect 4 ename ZeroDivisionError
nb expect 4 evalue "division by zero"
nb expect 4 tb "ZeroDivisionError: division by zero"
nb expect 5 src "raw text stays raw"
nb expect 6 val ""
nb run 6
nb expect 6 val 9
nb cell 4
# written by hand, dropped into NOTEBOOKS (Windows)
usb drop tests/notebooks/handmade.ipynb win as NOTEBOOKS/handmade.ipynb
wait 1000
expect state usb_result "Imported handmade"
wait 2000
nb open handmade
nb expect 1 src "Sources may be one string"
nb expect 2 src "√(16)+x^2"
nb run 2
ifcas src nb expect 2 val x^2+4
nb expect 3 src "x^2-1, -3, 3"
nb run 3
nb expect 3 plot x^2-1
nb expect 4 src "café"
nb run 4
nb expect 4 val 6
nb expect 5 src "an unknown cell type becomes text"
nb json
# broken files: the reason, nothing stored
usb drop tests/notebooks/broken.ipynb mac
wait 1000
expect state usb_result "Not a notebook: broken.ipynb"
wait 2000
usb expect RESULT.TXT "Nothing was installed."
usb expect RESULT.TXT "line 6"
usb drop tests/notebooks/old.ipynb win
wait 1000
expect state usb_result "Not a notebook: old.ipynb"
wait 2000
usb expect RESULT.TXT "nbformat 3 is not supported"
# replacing a notebook (the Finder's replace in NOTEBOOKS): the previous version is kept
usb drop build/ui_nb_algebra.ipynb mac replace as NOTEBOOKS/Algebra.ipynb
wait 1000
expect state usb_result "Imported Algebra"
wait 2000
usb expect RESULT.TXT "It replaced the notebook of that name"
nb revert Algebra
nb open Algebra
ifcas src nb expect 1 val x^3/3
nb expect 3 plot sin(x)
# while the app is open a dropped notebook waits; leaving it imports it
app notebook
usb drop "tests/notebooks/From Jupyter.ipynb" as "Second.ipynb"
wait 1000
expect state usb_pending Second.ipynb
expect state toast "Leave Notebook to import it"
key HOME
wait 200
expect state usb_result "Imported Second"
wait 2000
usb expect NOTEBOOKS/Second.ipynb "Imported from Jupyter"
# a new notebook (the list's F2), a calculation in it, saved: the drive is generated again (the
# store changed) and lists it
app notebook
expect state nb_view list
key F2
key CLEAR
type "Fresh"
key ENTER
expect screen home
type "1+1\n"
nb flush
wait 3000
usb expect NOTEBOOKS/Fresh.ipynb "\"text/plain\": "
usb ls NOTEBOOKS
usb host off
