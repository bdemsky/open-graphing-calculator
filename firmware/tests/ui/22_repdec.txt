# Repeating decimals: 0.(3) input is exact, decimal results with a short period show the period
expect state repeat 1
# input: the period in parentheses
type "0.(3)\n"
expect state last_output 1/3
type "1.2(34)\n"
expect state last_output 611/495
type "0.(9)\n"
expect state last_output 1
type "2*0.1(6)\n"
expect state last_output 1/3
