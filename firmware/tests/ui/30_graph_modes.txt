# Graph modes: Y= header and MODE row, parametric, polar, sequences (time and
# web), implicit relations with inequality shading, per-mode WINDOW and TABLE,
# saved across power off/on.
key F1
expect state yeq_mode FUNC
# UP from Y1 reaches the mode header; RIGHT/ENTER choose PAR
key UP
expect state yeq_row -1
key RIGHT ENTER
expect state graph_type PAR
expect text X1T
# X,T,θ,n types t in PAR mode
# ENTER on a mode puts the focus on its first entry
expect state yeq_row 0
type "3cos(x)"
key DOWN
type "2sin(x)"
expect state e1_1 3cos(t)
expect state e1_2 2sin(t)
screenshot par_yeq.png
key F5
expect state graph_plotted 1
expect state graph_error ""
# the ellipse is drawn in blue
expect color 0x029b 300
screenshot par_graph.png
# TRACE steps T by Tstep (π/24)
key F4
expect state trace_t 0
expect state trace_x 3
key RIGHT RIGHT
expect state trace_t 0.2617993878
expect text T=0.26179939
# WINDOW has Tmin/Tmax/Tstep first
key F2
expect text Tstep=
key DOWN DOWN
type "pi/12"
key ENTER
expect statecontains screen window
key F5 F4 RIGHT
expect state trace_t 0.5235987756
# TABLE lists T, X1T, Y1T
key F1 SHIFT F5
expect text X1T
expect text Y1T
expect text 1.62090692
key HOME
# --- polar: MODE row GRAPH
setmode GRAPH POL

key F1
expect text r1

key F5
expect state graph_plotted 1
expect color 0x029b 200
screenshot pol_graph.png
key F4 RIGHT RIGHT RIGHT
expect statecontains trace_t 0.3926990
expect text R=

type "1"
key F5

key F1
type ",1"
key F5

expect state graph_plotted 2

key DOWN RIGHT RIGHT RIGHT

expect state trace_y 8

# web plot: needs u(n-1) only; the cobweb of u1 = 1.5u1(n-1)
