# STAT TESTS (tests, intervals, Draw), DISTR menu and DRAW, distributions on
# the home screen and in Y=
app stat
key 3
expect state menu "STAT TESTS"
screenshot stat_tests_menu.png
# Z-Test with summary statistics: mu0=3, sigma=2, xbar=3.8, n=5
key 1
expect screen statform
expect text Z-Test
key RIGHT
expect text xbar:
key DOWN
type "3\n2\n3.8\n5\n"
key DOWN ENTER
expect screen statresult
expect text z=0.894427191
expect text p=0.371093369523
screenshot stat_ztest.png
# Draw: the two tails are shaded
key ENTER
expect screen statdraw
expect text p=0.371093369523
expect color 0x655F 1000
screenshot stat_ztest_draw.png
key CLEAR
expect screen statresult
key CLEAR
expect screen statform
key CLEAR
# T-Test from data, one-sided
stat set L1 5.1,4.9,5.6,5.8,6,5.2,4.7,5.5
app stat
key 3 2
key DOWN
type "5\n"
key DOWN DOWN RIGHT RIGHT DOWN ENTER
expect text t=2.19795038968
expect text p=0.0319620614694
expect text df=7
key CLEAR CLEAR
# 2-SampTTest (Welch) from L1, L2
stat set L2 4.2,4.8,5,4.4,4.6,5.1,4.3
app stat
key 3 4
key UP UP ENTER
expect text t=3.48552271527
expect text df=12.8523098071
key CLEAR CLEAR
# 1-PropZTest and its Draw (upper tail)

# chi² GOF: contributions in CNTRB
stat set L1 18,22,30,30
stat set L2 25,25,25,25
app stat
key 3 7
key DOWN DOWN
type "3\n"
key ENTER
expect text chi²=4.32
expect text p=0.228918864336
stat expect CNTRB 1.96,0.36,1,1
key CLEAR CLEAR
