# 3D graphing: contours (help page graph3d-contours). VIEW menu "Contours" draws the level curves on
# the z = f(x,y) surfaces in the 3D view; "Contour map" shows them flat in the x-y plane, filled
# with the height colours (shaded style) or as lines (wireframe), with a key of the level values;
# the number of levels is the 3D window's "levels"; parametric surfaces and space curves get none;
# the settings survive a power cycle and travel with a 3D graph cell. Started away from the home
# screen, like tests/ui/35_graph3d.txt.
app settings
app graph3d
expect state g3d_contours 0
expect state g3d_map 0
expect state g3d_levels 9
key F1
type "x^2+y^2"
key F5
expect screen graph3d
expect state g3d_contour_segs 0
# VIEW menu: Contours on (item 6); zfit gives z 0..50: nine circles
key F3
expect screen menu
expect state menu "3D VIEW"
key 6
expect screen graph3d
expect state g3d_contours 1
key SHIFT F4
expect state g3d_window -5,5,-5,5,0,50
expect state g3d_contour_segs 798
screenshot g3d_contours_bowl.png
# colour by height, wireframe: the curves alone in the height colours
key SHIFT F3 SHIFT F2
expect state g3d_style wire
expect state g3d_contour_segs 798
screenshot g3d_contours_wire.png
key SHIFT F2 SHIFT F3
# the contour map (item 5): filled bands, frame values, the key of the levels
key F3 5
expect state g3d_map 1
expect state g3d_contour_segs 476
expect text "z"
expect text "45"
screenshot g3d_contour_map_bowl.png
# trace works in the map; an arrow then moves the cursor, not the view
key F4
expect state g3d_trace "Z1 x=0 y=0 z=0"
key RIGHT RIGHT RIGHT
expect state g3d_trace "Z1 x=1.25 y=0 z=1.5625"
expect state g3d_map 1
screenshot g3d_contour_map_trace.png
key CLEAR
# turning leaves the map: back to the 3D view, turned from where it was
key RIGHT
expect state g3d_map 0
expect state g3d_azim -45
key F3 1
expect state g3d_azim -60
# the number of levels: the 3D window's last field (UP from the first field wraps to it)
key F2
expect screen graph3d_win

type "40"
key ENTER
expect text "ERR: levels must be 1 to 30"
key CLEAR
type "4"
key ENTER
expect state g3d_levels 4
key F5
expect screen graph3d
expect state g3d_contour_segs 360
key F2 UP
type "9"
key ENTER
