# Solver app: equation solver, polynomial roots, linear system; saved state
app solver
expect state menu SOLVER
key 1
expect screen eqsolver
type "x^2+y=7"
key ENTER
expect text y=
key DOWN
type "3\n"
key UP UP
# ALPHA ENTER solves for the highlighted variable, nearest to the guess 0
key ALPHA ENTER
expect text "x solved"
expect text "root1=-2"
expect text "root2=2"
screenshot solver_eq.png
# a new guess picks the other root
type "5"
key SHIFT ENTER
expect text "left-right=0"
key DOWN
expect text "x=2"
# solving for y
key ALPHA ENTER
expect text "y solved"
# bounds limit the search
key DOWN
type "0\n"
key UP UP UP
type "~5\n"
key UP
key ALPHA ENTER
expect text "1 root"
# no solution
key UP
type "x^2+1\n"
key ALPHA ENTER
expect text "ERR: no root found"
key HOME
# polynomial roots
app solver
key 2
expect screen polyroot
type "3\n1\n0\n0\n~8\n"
expect text "-1-1.732050808i"
expect text "-1+1.732050808i"
expect text "x3="
screenshot solver_poly.png
key HOME
# linear system: 2 equations, 3 unknowns -> two parameters
app solver
key 3
expect screen linsys
key UP RIGHT
type "4"
key ENTER
type "1\n2\n3\n6\n2\n4\n6\n12\n"
key ALPHA ENTER
expect screen linres
expect text "infinitely many (2 parameters)"
expect text "x1=6-2t1-3t2"
screenshot solver_lin.png
key CLEAR
expect screen linsys
# unique solution: x+y=3, x-y=1
key UP UP UP
type "2"
key RIGHT
type "3"
key DOWN
type "1\n1\n3\n1\n~1\n1\n"
key ALPHA ENTER
expect text "unique solution"
expect text "x1=2"
expect text "x2=1"
key CLEAR
# inconsistent
key UP UP UP
key DOWN
type "1\n1\n3\n1\n1\n4\n"
key ALPHA ENTER
expect text "no solution"
key HOME
# everything is kept across power off/on
key SHIFT ON
key ON
app solver
key 1

app solver
