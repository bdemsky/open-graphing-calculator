# RPN levels are CAS values (hidden variables _rpnN), not printed text:
# uncertainty sources stay correlated, floats keep the digits the display
# rounds away, exact values stay exact; stack operations, LASTARG, UNDO, a
# power cycle and a backup keep the values; dropped levels leave no hidden
# CAS variables behind.
setmode ENTRY RPN
key HOME
# x ENTER - of a value with an uncertainty is 0±0 (both levels are one value)
ifcas src type "2 0.1 pm\n"
ifcas src expect state rpn_1 2.00±0.10
ifcas src key ENTER SUB
ifcas src expect state rpn_1 0±0
# x ENTER × is correlated like x*x (σ = 2|x|s), not like two measurements
ifcas src type "3 0.2 pm\n"
ifcas src key ENTER MUL
ifcas src expect state rpn_1 9.0±1.2
ifcas src expect eval "(3±0.2)*(3±0.2)" "9.0±0.8"
ifcas src expect eval "x:=3±0.2" "3.0±0.2"
ifcas src expect eval "x*x" "9.0±1.2"
ifcas src expect eval "purge(x)" "true"
