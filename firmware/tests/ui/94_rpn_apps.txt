# Integration: RPN entry mode with every app. Each app is opened from the RPN
# home screen and CLEAR / HOME return to it; values move between the stack
# and the apps through the clipboard. Bug: MENU F3 on the RPN screen copied
# only the command line (nothing with an empty line or a selected level).
setmode ENTRY RPN
key HOME
expect screen rpn
type "2\n3\n"
key MUL
expect state rpn_1 6
# copy: level 1 with an empty line, the selected level in the interactive stack
key MENU F3
expect state clip 6
type "10\n"
key UP UP
expect state rpn_sel 2
key MENU F3
expect state clip 6
key CLEAR
expect state rpn_sel 0
type "7"
key MENU F3
expect state clip 7
key CLEAR
# spreadsheet: paste a stack value into a cell, copy the result back
app sheet
sheet select A1
key MENU F4 ENTER
sheet value A1 7
sheet set A2 =A1*5
sheet select A2
key MENU F3
key CLEAR
expect screen rpn
key MENU F4 ENTER
expect state rpn_1 35
# statistics: a list from the stack's values, CLEAR returns to RPN
stat set L1 1,2,3
app stat
key 1
expect screen statedit
key CLEAR
expect screen rpn
key HOME
expect screen rpn

app graph

# the UNITS menu types into the command line
key SHIFT VARS DOWN DOWN ENTER DOWN DOWN DOWN ENTER
expect state rpn_line _km
key CLEAR
type "4"
key SHIFT VARS 1
type "0.5"
ifcas src key ENTER
ifcas src expect state rpn_1 4.0±0.5
# Phase 2 command names as RPN words (bug: nominal, uncert, pm, ucov were
# pushed as names): uncert of 4.0±0.5, 3 0.2 pm, nominal
ifcas src type "uncert\n"
ifcas src expect state rpn_1 0.5
ifcas src type "3 0.2 pm\n"
ifcas src expect state rpn_1 3.0±0.2
ifcas src type "nominal\n"
ifcas src expect state rpn_1 3.0
ifcas src expect state rpn_2 0.5
ifcas src key DEL
ifcas src expect state rpn_1 0.5
