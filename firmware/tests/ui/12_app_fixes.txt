# Regression tests for the app framework, home, menus, forms and help
# (message box, session save retries, MEM reset settings, ALPHA menu items,
# console eval, history overflow, WINDOW fields, About back link, VARS)
#
# `expect state KEY` without a value compares with "" (no stray argv[3])
expect state session_dropped
# message box: a 5-line text leaves out the "press any key" hint (it would
# overlap the 5th line); a line of 26 two-byte characters is drawn whole
msg "LONG" "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123abcdefghijklmnopqrstuvwxyz"
expect screen message
expect text 0123
expect notext "press any key"
screenshot msg_four_lines.png
key ENTER
msg "SHORT" "two lines of text: abcdefghijklmnop"
expect text "press any key"
key ENTER
msg "GREEK" "θθθθθθθθθθθθθθθθθθθθθθθθθθ"
expect text "θθθθθθθθθθθθθθθθθθθθθθθθθθ"
key ENTER
# session: a failed autosave stays pending and is retried after the next idle period
type "1+1\n"
session fail 1
wait 61000
expect state session_rc -1
expect state session_dirty 1
wait 61000
expect state session_rc 0
expect state session_dirty 0
# every blob that does not fit is reported, not only the last one
session limit 64
session save
expect statecontains session_dropped graph,textbook,theme
expect statecontains session_dropped home,cas
session limit 0
session save
expect state session_dropped ""
