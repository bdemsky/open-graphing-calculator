# Undo is per cell in the editors that edit one cell at a time (list and
# matrix editors): moving to another cell forgets the steps, so MENU F1 never
# brings back the previous cell's text (app/input_undo.c input_undo_new_field)
app stat
key 1
expect screen statedit
type "123\n"
stat expect L1 123
expect state statedit_row 2
# a new cell: nothing to undo (the old bug pasted 123 back)
key MENU F1
expect screen statedit
key ENTER
stat expect L1 123
# inside a cell undo still works
type "45"
key MENU F1
type "6\n"
stat expect L1 123,6
# editing a value: undo stops at the loaded value
key UP ENTER
type "7"
key MENU F1 MENU F1
key ENTER
stat expect L1 123,6
key HOME
# matrix editor (the real CAS: matrices are CAS variables)
ifcas src key SHIFT MATH 3
ifcas src key ENTER
ifcas src expect screen matedit
ifcas src key UP
ifcas src type "2\n2\n"
ifcas src type "9\n"
ifcas src key MENU F1
ifcas src expect screen matedit
ifcas src key ENTER
ifcas src type "34"
ifcas src key MENU F1
ifcas src type "5\n"
ifcas src key HOME
ifcas src type "matA\n"
ifcas src expect state last_output [[9,0],[5,0]]
