# --- equation solver: F5 on the eq: line solves for its only variable;
# every root on its own row; F5 on a root row uses that root
app solver
key 1
type "x^3-2x=1"
expect text "type an equation, e.g. x^3-2x=1, then ENTER"
key ENTER
expect text "F5: solve for x (its value is the guess)"
key UP
key F5
expect text "x solved: 3 roots, left-right=0"
expect text "root1=-1"
expect text "root2=-0.6180339887"
expect text "root3=1.618033989"
expect state screen eqsolver
screenshot solver_roots.png
key DOWN DOWN DOWN DOWN DOWN
expect text "F5: use this root as x"
key F5
expect text "x=1.618033989 (root 3 of 3)"
expect text "x=1.618033989"
# two variables: F5 on the eq: line asks for the unknown
key UP
type "x+y=3\n"
key UP
key F5
expect text "Select the unknown"
