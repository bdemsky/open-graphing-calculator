# CALC menu (SHIFT F4 on the graph screens): value, zero, minimum, maximum, intersect, dy/dx,
# integral (shaded), inflection, arc length, table at the cursor. Results are
# shown on the graph and stored in ans.
key F1
type "x^2-2"
key DOWN
type "sin(x)"
key F5 ENTER
key SHIFT F4
expect screen menu
expect state menu CALCULATE
key SHIFT MENU
expect state help_topic graph-calc
key CLEAR
expect state menu CALCULATE
# zero of Y1: left bound by cursor keys, right bound typed, guess at the cursor
key 2
expect screen graph
expect state calc_prompt "Left bound?"
expect state trace_fn 1
key LEFT LEFT LEFT LEFT LEFT LEFT LEFT LEFT LEFT LEFT
key ENTER
expect state calc_prompt "Right bound?"
type "3"
key ENTER
expect state calc_prompt "Guess?"
key ENTER
expect state calc_prompt ""
expect statecontains graph_result Zero  X=1.41421356
expect state graph_label "Zero (1.414214, 0)"
expect eval "ans" "1.41421356237"
screenshot calc_zero.png
# minimum of Y1 over [-2, 2]
key SHIFT F4 3
type "~2"
key ENTER
type "2"
key ENTER ENTER
expect statecontains graph_result Minimum  X=0
expect statecontains graph_result Y=-2
# maximum of sin over [0, 3]: switch the curve with DOWN
key SHIFT F4 4
key DOWN
expect state trace_fn 2
type "0"
key ENTER
type "3"
key ENTER ENTER
expect statecontains graph_result Maximum  X=1.57079633
expect statecontains graph_result Y=1
# intersection of Y1 and Y2 near x = 1.5: with exactly two curves there is
# nothing to choose, so it asks for the guess at once (curve choice: 36_graph_usability)
key SHIFT F4 5
expect state calc_prompt "Guess?"
expect text "Guess?  Y1 and Y2"
type "1.5"
key ENTER
expect statecontains graph_result Intersection  X=1.72846632
screenshot calc_isect.png
