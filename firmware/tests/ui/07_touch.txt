# Touch input (capacitive panel): scripted contacts go through the same gesture
# recognizer as the FT6236 driver. Everything here also works by keys.
expect state touch_available 1
# --- menus: tap an item to choose it (CAS menu rows start at y=42, 20 px each)
key CAS
expect screen menu
touch tap 60 52
expect screen home
expect state entry factor()
expect state touch_last "tap 60,52"
key CLEAR
# drag up scrolls a long list (CATALOG), then a tap chooses the visible item
key SHIFT APPS
expect state menu CATALOG
touch drag 100 200 100 80 6
expect state menu_sel 6
touch tap 100 52
expect screen home
key CLEAR CLEAR
# --- home history: tap selects an entry, a second tap pastes it
type "6*7\n"
type "1+1\n"
touch tap 20 28
expect state home_sel 0
touch tap 20 28
expect state home_sel -1
expect state entry 6*7
# tap on the entry line moves the cursor to the tapped column
touch tap 15 108
expect state cursor 1
key DEL
expect state entry *7
# a tap in the empty area deselects
key CLEAR UP
expect state home_sel 3
touch tap 160 200
expect state home_sel -1
key CLEAR CLEAR
# --- Y=: tap '=' toggles, tap the swatch cycles the colour, tap the row edits it
# (the mode header is the first 22 px of the body, Y1 starts at y=40, 30 px rows)
key F1
type "x^2"
touch tap 36 55
expect state y1_on 0
touch tap 36 55
expect state y1_on 1
touch tap 5 55
expect state y1_color red
touch tap 5 55
expect state y1_color black
touch tap 5 55
touch tap 5 55
touch tap 5 55
touch tap 5 55
touch tap 5 55
touch tap 5 55
expect state y1_color blue
touch tap 120 85
type "3"
expect state y2 3
key CLEAR
expect state y2 ""
# --- graph: tap places the trace cursor on the nearest function (graph: a cell; ENTER: the graph app on it)
key F5 ENTER
expect screen graph
touch tap 160 129
expect state graph_mode trace
expect state trace_fn 1
expect state trace_x 0.03134796238
# drag pans: 100 px right moves the window left by 100/319 of its width
touch drag 60 120 160 120
expect state xmin -16.26959248
expect state xmax 3.730407524
expect state graph_mode trace
# pinch out (fingers 60 -> 120 px apart) zooms in 2x about the point between
# the fingers (pixel 160 is x = 0.0313 in ZStandard, so the window is not symmetric)
key F3 3
touch pinch 160 129 60 120
expect state xmin -4.984326019
expect state xmax 5.015673981
screenshot touch_graph_zoomed.png
lcdcheck
# pinch in zooms out again
touch pinch 160 129 120 60
expect state xmin -10
expect state xmax 10
# single contacts with explicit states; a second finger cancels a drag preview
touch 100 100 down
touch 140 100 move
touch 200 100 down 1
expect state xmin -10
touch 200 100 up 1
touch 140 100 up
expect state xmin -10
lcdcheck
# --- MODE: tap an option

pattern bars

key SHIFT ON
