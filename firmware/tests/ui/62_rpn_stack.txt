# RPN interactive stack (▲), soft keys, editing levels, counted commands,
# long values, scrolling and touch
setmode ENTRY RPN
key HOME
type "10\n20\n30\n"
type "40\n50\n"
# ▲ selects level 1, ▲ moves up; soft keys at the bottom
key UP
expect state rpn_sel 1
expect text echo
expect text rolld
key UP UP
expect state rpn_sel 3
# 3 ROLL: level 3 to level 1
key 3
expect state rpn_sel 0
expect state rpn_1 30
expect state rpn_2 50
expect state rpn_3 40
# F4 (rolld; also the digit 4): level 1 to level n
key UP UP UP UP
expect state rpn_sel 4
key F4
expect state rpn_1 50
expect state rpn_4 30
# 2 PICK copies level n
key UP UP UP UP UP
key 2
expect state rpn_depth 6
expect state rpn_1 10
# ENTER (ECHO) copies the selected level into the command line
key UP UP
key ENTER
expect state rpn_line 50
expect state rpn_sel 0
key CLEAR
# DEL in the interactive stack drops only the selected level
key UP UP DEL
expect state rpn_depth 5
expect state rpn_sel 2
expect state rpn_2 40
# 5 DROPN drops levels 1..n
key 5
expect state rpn_depth 3
expect state rpn_1 20
# ▼ below level 1 leaves the interactive stack
key UP DOWN
expect state rpn_sel 0
# 6 EDIT (and ▼ on an empty line for level 1) edits a level in place
key UP UP 6
expect state rpn_edit 2
expect state rpn_line 30
key LEFT LEFT DEL
type "7"
key ENTER
expect state rpn_edit 0
expect state rpn_2 70
expect state rpn_depth 3
key DOWN
expect state rpn_edit 1
key CLEAR
expect state rpn_edit 0
expect state rpn_depth 3
# counted commands: n in level 1 (ROLL, PICK, DROPN, DUPN)
key CLEAR CLEAR
type "1 2 3 4 3\n"
type "roll\n"
expect state rpn_1 2
expect state rpn_2 4
type "4 pick\n"
expect state rpn_1 1
type "2 dupn\n"
expect state rpn_depth 7
expect state rpn_1 1
expect state rpn_2 2
type "6 dropn\n"
expect state rpn_depth 1
type "7 dropn\n"
expect state rpn_error "Bad argument value"
# the STACK menu: 1 ENTER ROLLD from the menu (rolld with n = 1 is a no-op)
key CLEAR CLEAR
type "5\n6\n7\n2\n"
key COMMA DOWN DOWN DOWN DOWN DOWN DOWN ENTER
expect state rpn_depth 3
expect state rpn_1 6
expect state rpn_2 7
# long values wrap onto several lines
key CLEAR CLEAR
ifcas src type "expand((x+1)^9)\n"
ifcas src expect statecontains rpn_1 x^9+9*x^8
ifcas src expect text "x^9+9*x^8"
ifcas src screenshot rpn_long.png
# many levels: the view scrolls with the selection, and by dragging
key CLEAR CLEAR
type "1\n2\n3\n4\n5\n6\n"
type "7\n8\n9\n10\n11\n"
type "12\n13\n14\n"
expect state rpn_depth 14
expect notext 14:
key UP UP UP UP UP UP UP UP UP UP UP UP
expect state rpn_sel 12
expect text 12:
expect state rpn_scroll 3
key CLEAR
expect state rpn_scroll 0
touch drag 100 60 100 160 8
expect state rpn_scroll 5
screenshot rpn_scrolled.png
touch drag 100 160 100 20 8
expect state rpn_scroll 0
# touch: tap a level to select it, tap again to copy it to the command line
touch tap 200 170
expect state rpn_sel 2
touch tap 200 170
expect state rpn_sel 0
expect state rpn_line 13
key CLEAR
# soft keys can be tapped (4 ROLLD)
touch tap 200 130
expect state rpn_sel 4
touch tap 200 232
expect state rpn_sel 0
expect state rpn_4 14
expect state rpn_1 13
